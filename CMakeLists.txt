cmake_minimum_required(VERSION 3.15)
project(variant14 LANGUAGES CXX)

option(USE_MAX_OPTIMIZATION "Use maximum optimization" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if (NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

if (MSVC)
    add_compile_options(/W3 /utf-8)
    if (USE_MAX_OPTIMIZATION)
        set(CMAKE_CXX_FLAGS_RELEASE "/O2 /DNDEBUG")
    else()
        set(CMAKE_CXX_FLAGS_RELEASE "/Od /Ob0 /DNDEBUG")
    endif()
else()
    add_compile_options(-Wall)
    if (USE_MAX_OPTIMIZATION)
        set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")
    else()
        set(CMAKE_CXX_FLAGS_RELEASE "-O0 -DNDEBUG")
    endif()
endif()

add_executable(variant14 main.cpp algorithms.cpp variant14.h)

if (USE_MAX_OPTIMIZATION)
    target_compile_definitions(variant14 PRIVATE OPT_NAME="O3")
else()
    target_compile_definitions(variant14 PRIVATE OPT_NAME="O0")
endif()

find_package(Threads REQUIRED)
target_link_libraries(variant14 PRIVATE Threads::Threads)

if (NOT MSVC)
    find_package(TBB QUIET)
    if (TBB_FOUND)
        target_link_libraries(variant14 PRIVATE TBB::tbb)
        target_compile_definitions(variant14 PRIVATE HAS_TBB=1)
    else()
        message(WARNING "TBB not found: par/par_unseq will run sequentially")
    endif()
endif()
